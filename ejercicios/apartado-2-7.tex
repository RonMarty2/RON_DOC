\apartado{2.7}{Variables aleatorias}

\noindent\textit{Hasta acá preguntábamos por eventos sueltos. Ahora se
describe el comportamiento completo de una variable, y se lo resume en dos
números.}
\recordar{%
  \textbf{Variable aleatoria discreta}: asigna un número a cada resultado del
  experimento, y toma valores separados (contables).\\[0.3em]
  $E(X)=\sum x_i\,P(x_i)$ \quad — el valor promedio a la larga.\\
  $\operatorname{Var}(X)=\sum (x_i-\mu)^2\,P(x_i)$ \quad — cuánto se dispersa.\\
  $\sigma = \sqrt{\operatorname{Var}(X)}$, en las mismas unidades que $X$.}

\ejercicio{0}{%
  ¿Cuáles de estas son variables aleatorias \emph{discretas} y cuáles
  \emph{continuas}?
  \begin{enumerate}[label=(\alph*),topsep=2pt,itemsep=1pt]
    \item El puntaje del cuestionario (0 a 27).
    \item El tiempo que tarda alguien en completarlo.
    \item La cantidad de estudiantes que dan positivo en una semana.
  \end{enumerate}}
\espacioresolver[2.2cm]
\respuesta{(a) discreta; (b) continua; (c) discreta.}
\solucion{%
  (a) \textbf{Discreta}: sólo puede valer $0,1,2,\ldots,27$. No existe el
  puntaje $14{,}3$.
  \\[0.3em]
  (b) \textbf{Continua}: entre 4 y 5 minutos hay infinitos valores posibles.
  \\[0.3em]
  (c) \textbf{Discreta}: se cuentan personas, y no hay media persona.
  \\[0.3em]
  La distinción importa porque cambian las herramientas: las discretas usan
  binomial, Poisson o hipergeométrica (2.8); las continuas, la normal (2.9).}

\ejercicio{1}{%
  Sea $X$ el resultado de tirar un dado. Escribí la función de masa, o sea la
  tabla de todos los valores posibles con su probabilidad. Verificá que las
  probabilidades suman 1.}
\espacioresolver[2.6cm]
\respuesta{$P(x)=1/6$ para $x=1,\ldots,6$; suman $6\times 1/6 = 1$.}
\solucion{%
  \begin{center}\small
  \begin{tabular}{@{}lcccccc@{}}
  \toprule
  $x$ & 1 & 2 & 3 & 4 & 5 & 6\\
  $P(x)$ & $1/6$ & $1/6$ & $1/6$ & $1/6$ & $1/6$ & $1/6$\\
  \bottomrule
  \end{tabular}
  \end{center}
  Suma: $6\times\tfrac{1}{6}=1$. \checkmark
  \\[0.3em]
  Que sumen 1 no es un detalle: es el segundo axioma. Si una tabla de
  probabilidades no suma 1, está mal.}

\ejercicio{1}{%
  Con la misma variable, calculá $E(X)$ desarrollando la suma término a
  término.}
\espacioresolver[2.4cm]
\respuesta{$E(X)=3{,}5$.}
\solucion{%
  \[ E(X)=1\cdot\tfrac{1}{6}+2\cdot\tfrac{1}{6}+3\cdot\tfrac{1}{6}
     +4\cdot\tfrac{1}{6}+5\cdot\tfrac{1}{6}+6\cdot\tfrac{1}{6}
     =\frac{21}{6}=3{,}5 \]}

\ejercicio{2}{%
  Calculá la varianza y la desviación estándar del resultado de un dado,
  sabiendo que $\mu=3{,}5$.}
\espacioresolver[3.4cm]
\respuesta{$\operatorname{Var}(X)\approx 2{,}92$; $\sigma\approx 1{,}71$.}
\solucion{%
  \[ \operatorname{Var}(X)=\frac{(1-3{,}5)^2+(2-3{,}5)^2+\cdots+(6-3{,}5)^2}{6} \]
  Los cuadrados son $6{,}25;\ 2{,}25;\ 0{,}25;\ 0{,}25;\ 2{,}25;\ 6{,}25$, que
  suman $17{,}5$.
  \[ \operatorname{Var}(X)=\frac{17{,}5}{6}\approx 2{,}92
     \qquad \sigma=\sqrt{2{,}92}\approx 1{,}71 \]
  \\[0.3em]
  La varianza está en «puntos al cuadrado», que no significa nada; por eso se
  informa $\sigma$, que vuelve a las unidades originales.}

\ejercicio{2}{%
  Una variable $X$ toma los valores 0, 1, 2 y 3 con probabilidades
  $0{,}4$; $0{,}3$; $0{,}2$ y $0{,}1$.
  \begin{enumerate}[label=(\alph*),topsep=2pt,itemsep=1pt]
    \item Verificá que es una distribución válida.
    \item Calculá $E(X)$.
  \end{enumerate}}
\espacioresolver[2.8cm]
\respuesta{(a) Suman 1. (b) $E(X)=1{,}0$.}
\solucion{%
  (a) $0{,}4+0{,}3+0{,}2+0{,}1 = 1$, y ninguna es negativa. \checkmark
  \\[0.3em]
  (b) $E(X)=0(0{,}4)+1(0{,}3)+2(0{,}2)+3(0{,}1)=0+0{,}3+0{,}4+0{,}3=1{,}0$.
  \\[0.3em]
  Ésta es exactamente la forma de una pregunta del cuestionario, donde las
  cuatro opciones no son igualmente probables: mucha gente responde 0 y poca
  responde 3.}

\ejercicio{3}{%
  En las 200 fichas, el puntaje de depresión tiene media $\mu = 6{,}32$ y
  desviación estándar $\sigma = 4{,}64$.
  \begin{enumerate}[label=(\alph*),topsep=2pt,itemsep=1pt]
    \item ¿Qué significa cada número, en palabras y en puntos del cuestionario?
    \item El corte está en 10. ¿Está por encima o por debajo de $\mu+\sigma$?
  \end{enumerate}}
\espacioresolver[3.4cm]
\respuesta{(a) Promedio 6,32 puntos; dispersión típica 4,64 puntos.
(b) $\mu+\sigma = 10{,}96$, así que el corte está por debajo.}
\solucion{%
  (a) El puntaje promedio del archivo es de $6{,}32$ puntos. La desviación
  dice que lo típico es apartarse unos $4{,}64$ puntos de ese promedio: la
  mayoría de las fichas cae más o menos entre $1{,}7$ y $11$ puntos.
  \\[0.3em]
  (b) $\mu+\sigma = 6{,}32+4{,}64 = 10{,}96$. El corte de 10 está
  \textbf{por debajo}: no llega a una desviación estándar por encima del
  promedio.
  \\[0.3em]
  Eso ya anticipa algo: el corte no es un valor extremo, así que va a marcar a
  bastante gente. En el apartado 2.9 se calcula cuánta exactamente.}

\ejercicio{3}{%
  El servicio va a tamizar a 20 estudiantes nuevos. La probabilidad de que
  cada uno dé positivo es $0{,}215$.
  ¿Cuántos positivos espera encontrar? Usá $E(X)=n\cdot p$.}
\espacioresolver[2.2cm]
\respuesta{$20\times 0{,}215 = 4{,}3$ estudiantes.}
\solucion{%
  \[ E(X) = n\cdot p = 20 \times 0{,}215 = 4{,}3 \]
  \\[0.3em]
  Espera unos 4 o 5 positivos. El valor esperado sirve justamente para esto:
  planificar cuántas entrevistas reservar.}

\ejercicio{4}{%
  Un estudiante objeta:
  \begin{quote}\itshape
  «$E(X)=3{,}5$ para un dado no puede estar bien: un dado nunca saca 3,5.»
  \end{quote}
  ¿Tiene razón? Explicá qué significa realmente el valor esperado, y dá otro
  ejemplo donde $E(X)$ tampoco sea un valor posible.}
\espacioresolver[4.5cm]
\respuesta{No tiene razón. $E(X)$ es el promedio a la larga, no un resultado
posible. Ej.: $4{,}3$ positivos entre 20 estudiantes.}
\solucion{%
  \textbf{No tiene razón}, aunque la observación es buena.
  \\[0.3em]
  $E(X)$ no es «el resultado más probable» ni «un resultado posible»: es el
  \textbf{promedio al que tienden los resultados si el experimento se repite
  muchas veces}. Si tirás un dado 10\,000 veces y promediás, el número se
  acerca a 3,5.
  \\[0.3em]
  Otro ejemplo: los $4{,}3$ positivos esperados entre 20 estudiantes. Nunca
  vas a encontrar 4,3 personas — vas a encontrar 3, 4, 5 o 6. Pero si repetís
  el tamizaje muchas semanas, el promedio se acerca a 4,3.
  \\[0.3em]
  \textit{Es el mismo malentendido que «la familia argentina promedio tiene
  1,9 hijos». Nadie tiene 1,9 hijos, y sin embargo el número informa.}}
